[quote]
 Даже если спирт замерзнет, все равно его не брошу, потому что спирт хороший.

== Quick Start

=== BPMN Model: parsing

[source,go]
----
import bpmn "github.com/esivres/c2h5oh/pkg/bpmn_model"

// Read a BPMN file
bmi, err := bpmn.ReadFromFile("process.bpmn")

// Get all service tasks
for _, task := range bpmn.GetTypedElements[bpmn.ServiceTask](bmi.ModelInstance) {
    fmt.Printf("Task: %s, type: %s\n", task.GetId(), task.GetName())
}
----

=== BPMN Model: creating a process

[source,go]
----
import bpmn "github.com/esivres/c2h5oh/pkg/bpmn_model"

bmi := bpmn.CreateExecutableProcess("my-process").
    StartEvent("start").
    ServiceTask("call-api").Name("Call API").
        ZeebeJobType("api-worker").
        ZeebeInput("=requestData", "input").
        ZeebeOutput("=response", "result").
    EndEvent("end").
    Done()

// Save to file
bpmn.WriteToFile(bmi, "my-process.bpmn")
----

=== BPMN Model: AdHoc SubProcess (AI Agent)

[source,go]
----
bmi := bpmn.CreateExecutableProcess("ai-agent").
    StartEvent("start").
    AdHocSubProcess("agent").Name("AI Agent").
        ZeebeJobType("kamundarf:adhoc:v1").
        ZeebeAdHocConfig("toolCallResults",
            "={id: _meta.id, name: _meta.name, content: toolCallResult}").
        ZeebeInput("=userQuery", "userQuery").
    EndEvent("end").
    Done()
----

=== BPMN Assertions: testing processes

[source,go]
----
import bpmn_assertions "github.com/esivres/c2h5oh/pkg/bpmn_asserts"

// Wait for an event
bpmn_assertions.Events().
    ProcessInstanceCompleted(processInstanceKey).
    WaitOn(s.RecordStream, 30*time.Second)

// Fluent assertions
bpmn_assertions.ForProcessInstance(s.T(), s.RecordStream, processInstanceKey).
    WithContext(ctx).
    IsCompleted().
    HasPassedElement("serviceTask1").
    HasPassedElementsInOrder("start", "serviceTask1", "end").
    HasNoIncidents().
    HasVariableWithValue("result", "expected")
----

== Key design decisions

|===
| Java (Camunda/Zeebe) | Go (Rumunda)

| Interface + Impl class | Interface + struct embedding
| Abstract generic builders | Struct composition + Go generics
| DOM (org.w3c.dom) | etree (github.com/beevik/etree)
| Generics `<T extends X>` | Go generics `[T X]`
| Exceptions | error return values
| Maven multi-module | Single Go module
|===
